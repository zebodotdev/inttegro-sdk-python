389    async def page(self, payload: dict | None=None):
390        """
391        List executed payouts with page-based pagination.
392
393        Retrieves a paginated history of payouts that have been created. Use this to build
394        payout history views, reconcile bank deposits, or export financial reports. Results
395        include successful, failed, and pending payouts sorted by creation date (most recent
396        first).
397
398        Args:
399            payload: Pagination parameters including:
400                - page_number: Page index to fetch (1-10 inclusive, default: 1)
401                - page_size: Number of payouts per page (1-256, default: 20)
402
403        Returns:
404            domain object containing:
405                - page: Object with:
406                    - number: The page number returned
407                    - size: Number of payouts in this page
408                    - payouts: Array of payout objects with:
409                        - id: Payout ID
410                        - amount: Amount and currency
411                        - status: Current status
412                        - destination_id: Financial account that received funds
413                        - initiated_by: How payout was initiated (automatic or manual)
414                        - latest_attempt_id: Most recent execution attempt
415                        - latest_error: Error details if payout failed
416
417        Example:
418            ```python
419            # Get recent payouts
420            result = client.payouts.page({"page_number": 1, "page_size": 20})
421            page = result
422
423            for payout in page["payouts"]:
424                amount = payout["amount"]
425                print(f"{payout['id']}: {amount['value']} {amount['currency']} - {payout['status']}")
426
427            # Paginate through all payouts
428            for page_num in range(1, 11):
429                result = client.payouts.page({
430                    "page_number": page_num,
431                    "page_size": 50
432                })
433
434                page = result
435                if page["size"] == 0:
436                    break
437
438                payouts = page["payouts"]
439                print(f"Page {page_num}: {len(payouts)} payouts")
440
441            # Get large page of payouts
442            result = client.payouts.page({
443                "page_number": 1,
444                "page_size": 100
445            })
446
447            # Build reconciliation report
448            def reconcile_payouts():
449                result = client.payouts.page({
450                    "page_number": 1,
451                    "page_size": 256  # Maximum page size
452                })
453
454                total_paid = {}
455                for payout in result.payouts:
456                    if payout["status"] != "paid":
457                        continue
458
459                    amount = payout["amount"]
460                    currency = amount["currency"]
461                    value = amount["value"]
462
463                    if currency not in total_paid:
464                        total_paid[currency] = 0
465
466                    total_paid[currency] += value
467
468                print("Total payouts by currency:")
469                for currency, total in total_paid.items():
470                    print(f"  {currency.upper()}: {total}")
471            ```
472
473        See Also:
474            - settings(): View payout configuration
475            - https://studio.inttegro.com/api/payouts/page
476        """
477        return await self.http.post('/payouts/page', payload or {})

Payouts resource for configuring and managing transfers of funds from your balance.

Payouts move money from your Inttegro balance to your financial accounts (bank accounts, mobile money wallets, or Dosh). You can configure automatic weekly payouts or trigger manual payouts on demand. Set different destinations for different currencies to route funds appropriately.

The typical setup is:

  1. Connect financial accounts via the FinancialAccounts resource
  2. Set payout destinations (which financial account receives which currency)
  3. Configure automatic or manual payout schedule
  4. Monitor payouts via the page() method

See https://studio.inttegro.com/payouts for detailed guides.
